file(GLOB_RECURSE LIBENGINE_SOURCES CONFIGURE_DEPENDS *.cpp)

#
# SDL_GPU shader embedding (Phase 2 / 6 of the SDL3 + SDL_GPU migration).
#
# The HLSL sources in shaders/ are precompiled to DXIL (D3D12 / Windows) via
# dxc.exe; the GLSL sources are precompiled to SPIR-V (Vulkan / Linux+Android)
# via glslang. The resulting .dxil and .spv binaries are checked into the
# source tree. At configure time we read whichever binaries exist and embed
# them as C arrays into a generated header (composite_shaders_embedded.h) under
# the build directory. screen.cpp (the GPU render path) #includes that header
# and at runtime picks DXIL or SPIR-V based on what SDL_GPU reports as
# supported.
#
# If you regenerate any shader binary (see shaders/README.md), re-run CMake to
# refresh the embedded header — a plain incremental rebuild won't notice.
#
set(SHADER_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
# Generate the embedded header into the SOURCE tree (not the build dir) so the
# Android NDK build (which uses Android.mk and doesn't run our CMake) picks up
# the same header. The file is checked into git; it gets regenerated by every
# CMake configure on a desktop dev machine that has the relevant shader
# toolchain (.dxil from dxc, .spv from glslang). Android only consumes it.
set(SHADER_GEN_HEADER ${SHADER_SRC_DIR}/composite_shaders_embedded.h)

# Helper: read a precompiled shader binary and turn it into a C array literal.
# When the file is missing or empty (no toolchain available, fresh checkout),
# emit `0` so the resulting `uint8_t arr[] = { 0 }` is still legal C++ — the
# runtime detects the empty payload by checking the bytecode size against a
# minimum or using SDL_CreateGPUShader's error path.
function(_embed_shader_bytes input_file out_var)
    if(EXISTS "${input_file}")
        file(READ "${input_file}" hex_bytes HEX)
    else()
        set(hex_bytes "")
    endif()
    if(hex_bytes STREQUAL "")
        set(${out_var} "0" PARENT_SCOPE)
    else()
        string(REGEX REPLACE "(..)" "0x\\1, " c_array "${hex_bytes}")
        set(${out_var} "${c_array}" PARENT_SCOPE)
    endif()
endfunction()

# Force CMake to re-run when any shader binary changes so the embedded header
# stays in sync without a manual reconfigure.
set_property(
    DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${SHADER_SRC_DIR}/composite.vert.dxil
    ${SHADER_SRC_DIR}/composite.frag.dxil
    ${SHADER_SRC_DIR}/cursor.frag.dxil
    ${SHADER_SRC_DIR}/composite.vert.spv
    ${SHADER_SRC_DIR}/composite.frag.spv
    ${SHADER_SRC_DIR}/cursor.frag.spv
    )

_embed_shader_bytes(${SHADER_SRC_DIR}/composite.vert.dxil COMPOSITE_VERT_DXIL_C_ARRAY)
_embed_shader_bytes(${SHADER_SRC_DIR}/composite.frag.dxil COMPOSITE_FRAG_DXIL_C_ARRAY)
_embed_shader_bytes(${SHADER_SRC_DIR}/cursor.frag.dxil    CURSOR_FRAG_DXIL_C_ARRAY)
_embed_shader_bytes(${SHADER_SRC_DIR}/composite.vert.spv  COMPOSITE_VERT_SPV_C_ARRAY)
_embed_shader_bytes(${SHADER_SRC_DIR}/composite.frag.spv  COMPOSITE_FRAG_SPV_C_ARRAY)
_embed_shader_bytes(${SHADER_SRC_DIR}/cursor.frag.spv     CURSOR_FRAG_SPV_C_ARRAY)

configure_file(
    ${SHADER_SRC_DIR}/composite_shaders_embedded.h.in
    ${SHADER_GEN_HEADER}
    @ONLY
    )

add_compile_options("$<$<COMPILE_LANG_AND_ID:C,AppleClang,Clang,GNU>:${GNU_CC_WARN_OPTS}>")
add_compile_options("$<$<COMPILE_LANG_AND_ID:CXX,AppleClang,Clang,GNU>:${GNU_CXX_WARN_OPTS}>")
add_compile_options("$<$<OR:$<COMPILE_LANG_AND_ID:C,MSVC>,$<COMPILE_LANG_AND_ID:CXX,MSVC>>:${MSVC_CC_WARN_OPTS}>")

add_library(engine STATIC ${LIBENGINE_SOURCES})

target_include_directories(
    engine
    PRIVATE
    ${SHADER_SRC_DIR}
    )

target_compile_definitions(
	engine
	PRIVATE
	# MSVC: suppress deprecation warnings
	$<$<OR:$<COMPILE_LANG_AND_ID:C,MSVC>,$<COMPILE_LANG_AND_ID:CXX,MSVC>>:_CRT_SECURE_NO_WARNINGS>
	$<$<CONFIG:Debug>:WITH_DEBUG>
	$<$<BOOL:${ENABLE_IMAGE}>:WITH_IMAGE>
	$<$<BOOL:${MACOS_APP_BUNDLE}>:MACOS_APP_BUNDLE>
	)

target_include_directories(
	engine
	PUBLIC
	$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
	)

target_link_libraries(
	engine
	smacker
	SDL3::SDL3
	SDL3_mixer::SDL3_mixer
	$<$<BOOL:${ENABLE_IMAGE}>:SDL3_image::SDL3_image>
	$<$<BOOL:${ENABLE_IMAGE}>:PNG::PNG>
	Threads::Threads
	ZLIB::ZLIB
	)
